# Misc Internals

## Variadics

The standard C^4 variadics (Not the C ones usable via extern) are just a compiler-backed struct with fancy intrinsics. The struct looks like this:

```qc
struct qc_variadic {
    void** items;
    int count;
    int current_index;
}
```

That's it. There's no special LLVM. Just a struct with some functions.

## Inline ASM

Inline ASM compiles to LLVM Inline ASM. This file 

```qc
int main() {
    `inline(R"(
        mov rax, 1
        mov rdi, 1
        mov rsi, $0r
        mov rdx, 5
        syscall
    )", "Hello", "~{rax,rdi,rsi,rdx,r11,rcx,memory}");
    return 0;
}

```
(Using a argument because we want the pointer to Hello)
becomes this exact LLVM (at O2)
```llvm
@.str.44 = private constant [6 x i8] c"Hello\00"
define noundef i32 @main() local_unnamed_addr {
entry:
  tail call void asm sideeffect inteldialect "\0A        mov rax, 1\0A        mov rdi, 1\0A        mov rsi, $0\0A        mov rdx, 5\0A        syscall\0A    ", "r,~{rax},~{rdi},~{rsi},~{rdx},~{r11},~{rcx},~{memory}"(ptr nonnull @.str.44) #0
  ret i32 0
}
```

excluding `tail` and `sideeffect` which only exist for optimization, along with `local_unnamed_addr`, It's simple. The compiler creates a unnamed `asm` function (inline asm, llvm just does inline asm this way) and calls it, hence the `call void`. The next argument is the param types (so just one r for the r parameter) followed by comma and then the clober string, then the arguments to the asm function.
This would become this asm.
```asm
	.intel_syntax noprefix
	.file	"master_module"
	.text
	.globl	__user_entry                    # -- Begin function __user_entry
	.p2align	4
	.type	__user_entry,@function
__user_entry:                           # @__user_entry
	.cfi_startproc
# %bb.0:                                # %entry
	mov	r8d, offset .L.str.44
	#APP


	mov	rax, 1
	mov	rdi, 1
	mov	rsi, r8
	mov	rdx, 5
	syscall


	#NO_APP
	xor	eax, eax
	ret
.Lfunc_end0:
	.size	__user_entry, .Lfunc_end0-__user_entry
	.cfi_endproc
                                        # -- End function
	.globl	main                            # -- Begin function main
	.p2align	4
	.type	main,@function
main:                                   # @main
	.cfi_startproc
# %bb.0:                                # %entry
	mov	r8d, offset .L.str.44
	#APP


	mov	rax, 1
	mov	rdi, 1
	mov	rsi, r8
	mov	rdx, 5
	syscall


	#NO_APP
	xor	eax, eax
	ret
.Lfunc_end1:
	.size	main, .Lfunc_end1-main
	.cfi_endproc
                                        # -- End function
	.type	.L.str.44,@object               # @.str.44
	.section	.rodata,"a",@progbits
.L.str.44:
	.asciz	"Hello"
	.size	.L.str.44, 6

	.ident	"Debian clang version 21.1.8 (10)"
	.section	".note.GNU-stack","",@progbits
```
Yes, that is the exact llvm generated IR. Stripping the useless code and making it more readable it would be:
```asm
	.intel_syntax noprefix
	.file	"master_module"
	.text
	.globl	main                            # -- Begin function main
	.p2align	4
	.type	main,@function
main:                                   # @main
	.cfi_startproc
# %bb.0:                                # %entry ; comments generated by LLVM to show what became what
	mov	r8d, offset .L.str.44 ; moved because those `r` arguments are replaced with _registers_. It put the argument we used (in this case, "Hello") into whatever register the compiler found convenient
	#APP
	mov	rax, 1
	mov	rdi, 1
	mov	rsi, r8
	mov	rdx, 5
	syscall
	#NO_APP
	xor	eax, eax
	ret
.Lfunc_end1:
	.size	main, .Lfunc_end1-main
	.cfi_endproc
                                        # -- End function
	.type	.L.str.44,@object               # @.str.44
	.section	.rodata,"a",@progbits
.L.str.44:
	.asciz	"Hello" ; the string
	.size	.L.str.44, 6 ; it's length
```

## Labels and Jmp

This literally just creates basic blocks and br instructions, which create labels and jmp instructions.

```qc
int main() {
    ...do_stuff
    if (!x) {
    err:
        return 1;
    }
    ...
    if (!y) {
    cleanup_x:
        ...
        jmp err;
    }
    ...
    if (!z) {
    cleanup_y:
        ...
        jmp cleanup_x;
    }
    return 0;
}
```

Would become:

```llvm 
define i32 @main() {
    ...
    %if_cond = icmp eq ptr %x, null
    br i1 %if_cond, label %if.body, label %if.end
if.body:
    br label %err
err:
    ret i32 1
if.end:
    ...
    %if_cond2 = icmp eq ptr %y, null
    br i1 %if_cond2, label %if.body2, label %if.end2
if.body2:
    br label %cleanup_x
cleanup_x:
    ...
    br label %err
if.end2:
    ...
    %if_cond3 = icmp eq ptr %z, null
    br i1 %if_cond3, label %if.body3, label %if.end3
if.body3:
    br label %cleanup_y
cleanup_y:
    ...
    br label %cleanup_x
if.end3:
    ret i32 0
}
```

Which would become something like:

```asm
_start:
    ...
    mov rax, [x]
    cmp rax, 0
    jne .if.end
.if.body:
    jmp .err
.err:
    ; exit with code 1
.if.end:
    ...
    mov rax, [y]
    cmp rax, 0
    jne .if.end2
.if.body2:
    jmp .cleanup_x
.cleanup_x:
    ...
    jmp .err
.if.end2:
    ...
    mov rax, [z]
    cmp rax, 0
    jne .if.end3
.if.body3:
    jmp .cleanup_y
.cleanup_y:
    ...
    jmp .cleanup_x
.if.end3:
    ; exit with code 0
```
